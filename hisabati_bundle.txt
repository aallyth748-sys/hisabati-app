/// الحساب المحلي: اسم المستخدم وكلمة المرور تُحفظ داخل قاعدة البيانات المشفّرة.
/// كلمة المرور لا تُحفظ كنص، بل بصمة مشتقة منها مع ملح عشوائي.
class Auth {
  Auth._();

  static const minPassword = 4;
  static const minUsername = 2;

  /// حالة الدخول الحالية. تُعاد إلى false عند فتح التطبيق، فيُطلب تسجيل الدخول في كل مرة.
  static final ValueNotifier<bool> session = ValueNotifier(false);

  static Map<String, dynamic>? get _account {
    final raw = Store.setting<Object?>('account', null);
    if (raw is! Map) return null;
    return Map<String, dynamic>.from(raw);
  }

  static bool get hasAccount => _account != null;
  static bool get isLoggedIn => hasAccount && session.value;
  static String get username => (_account?['username'] as String?) ?? '';

  static Map<String, dynamic> _make(String username, String password) {
    final rnd = Random.secure();
    final salt = List<int>.generate(16, (_) => rnd.nextInt(256));
    return {
      'username': username,
      'salt': base64Encode(salt),
      'hash': base64Encode(deriveKey(password, salt)),
    };
  }

  static bool _verify(Map<String, dynamic> acc, String password) {
    final salt = base64Decode(acc['salt'] as String);
    final expected = base64Decode(acc['hash'] as String);
    final actual = deriveKey(password, salt);
    if (actual.length != expected.length) return false;
    var diff = 0;
    for (var i = 0; i < actual.length; i++) {
      diff |= actual[i] ^ expected[i];
    }
    return diff == 0;
  }

  static Future<void> createAccount(String username, String password) =>
      Store.setSetting('account', _make(username, password));

  /// يتحقق من البيانات ويفتح الجلسة. يعيد false إذا كانت البيانات خاطئة.
  static bool login(String username, String password) {
    final acc = _account;
    if (acc == null) return false;
    if (acc['username'] != username || !_verify(acc, password)) return false;
    session.value = true;
    return true;
  }

  static bool checkPassword(String password) {
    final acc = _account;
    return acc != null && _verify(acc, password);
  }

  static Future<void> changeUsername(String username) async {
    final acc = _account;
    if (acc == null) return;
    acc['username'] = username;
    await Store.setSetting('account', acc);
  }

  static Future<void> changePassword(String newPassword) async {
    final acc = _account;
    if (acc == null) return;
    final fresh = _make(acc['username'] as String, newPassword);
    await Store.setSetting('account', fresh);
  }

  static void logout() {
    session.value = false;
  }
}

/// النسخ الاحتياطي: ملف مشفّر بكلمة مرور يحدّدها المستخدم، ويُستعاد بها.
class Backup {
  Backup._();

  static const _format = 'hisabati-backup';

  static String _encrypt(String plain, String password) {
    final rnd = Random.secure();
    final salt = List<int>.generate(16, (_) => rnd.nextInt(256));
    final key = enc.Key(deriveKey(password, salt));
    final iv = enc.IV.fromSecureRandom(16);
    final encrypter = enc.Encrypter(enc.AES(key, mode: enc.AESMode.cbc));
    final encrypted = encrypter.encrypt(plain, iv: iv);
    return jsonEncode({
      'format': _format,
      'version': 1,
      'salt': base64Encode(salt),
      'iv': iv.base64,
      'data': encrypted.base64,
    });
  }

  /// يعيد النص الأصلي، ويرمي خطأ إذا كانت كلمة المرور خاطئة أو الملف تالفاً.
  static String decrypt(String fileText, String password) {
    final map = jsonDecode(fileText) as Map<String, dynamic>;
    if (map['format'] != _format) {
      throw const FormatException('ملف غير صالح');
    }
    final salt = base64Decode(map['salt'] as String);
    final key = enc.Key(deriveKey(password, salt));
    final iv = enc.IV.fromBase64(map['iv'] as String);
    final encrypter = enc.Encrypter(enc.AES(key, mode: enc.AESMode.cbc));
    return encrypter.decrypt64(map['data'] as String, iv: iv);
  }

  /// ينشئ ملف النسخة ثم يفتح قائمة المشاركة (حفظ أو إرسال).
  static Future<void> exportFile(String password) async {
    final text = _encrypt(jsonEncode(Store.exportData()), password);
    final dir = await getTemporaryDirectory();
    final file = File('${dir.path}/hisabati_${DateTime.now().millisecondsSinceEpoch}.hbak');
    await file.writeAsString(text);
    await Share.shareXFiles(
      [XFile(file.path)],
      text: 'نسخة احتياطية - البزي',
    );
  }

  /// يختار ملف النسخة من الهاتف ويعيد نصه، أو null إذا ألغى المستخدم.
  static Future<String?> pickFileText() async {
    final result = await FilePicker.platform.pickFiles(type: FileType.any);
    final path = result?.files.single.path;
    if (path == null) return null;
    return File(path).readAsString();
  }

Future<String> exportExcel({
  required String from,
  required String to,
  required String username,
}) async {
  bool inRange(String d) => d.compareTo(from) >= 0 && d.compareTo(to) <= 0;

  final rows = Ledger.build().where((r) => inRange(r.date)).toList();
  final entries = Store.all('entries').where((e) => inRange(e['date'] as String)).toList()
    ..sort((a, b) => (a['date'] as String).compareTo(b['date'] as String));
  final photos = Store.all('attachments').where((a) => inRange(a['date'] as String)).toList()
    ..sort((a, b) => (a['date'] as String).compareTo(b['date'] as String));

  final received = rows.fold<int>(0, (s, r) => s + r.received);
  final spent = rows.fold<int>(0, (s, r) => s + r.spent);
  final last = rows.isEmpty ? null : rows.last;

  final excel = Excel.createExcel();
  final sheet = excel['التقرير'];
  excel.delete('Sheet1');

  void addRow(List<Object> values, {bool bold = false}) {
    sheet.appendRow(values
        .map((v) => v is int ? IntCellValue(v) : TextCellValue(v.toString()))
        .toList());
    if (bold) {
      final r = sheet.maxRows - 1;
      for (var c = 0; c < values.length; c++) {
        sheet
            .cell(CellIndex.indexByColumnRow(columnIndex: c, rowIndex: r))
            .cellStyle = CellStyle(bold: true);
      }
    }
  }

  addRow(['البزي'], bold: true);
  addRow(['تقرير من ${fmtDate(parseYmd(from))} إلى ${fmtDate(parseYmd(to))}']);
  addRow(['']);

  for (final kind in spendingKinds) {
    final list = entries.where((e) => e['kind'] == kind).toList();
    addRow([kindLabels[kind]!], bold: true);
    addRow(['التاريخ', fieldLabels[kind]!, 'المبلغ', 'ملاحظة'], bold: true);
    for (final e in list) {
      addRow([
        fmtDate(parseYmd(e['date'] as String)),
        entryTitle(e),
        amountOf(e),
        (e['note'] as String?) ?? '',
      ]);
    }
    addRow(['الإجمالي', '', sumOf(list), ''], bold: true);
    addRow(['']);
  }

  addRow(['الاستلام'], bold: true);
  addRow(['التاريخ', 'المبلغ'], bold: true);
  for (final e in entries.where((e) => e['kind'] == kindReceived)) {
    addRow([fmtDate(parseYmd(e['date'] as String)), amountOf(e)]);
  }
  addRow(['']);

  addRow(['الملخص'], bold: true);
  addRow(['إجمالي المستلم', received]);
  addRow(['إجمالي المصروف', spent]);
  addRow([
    last == null || last.end >= 0 ? 'المتبقي' : 'المديونية',
    last == null ? 0 : last.end.abs(),
  ], bold: true);
  addRow(['']);

  addRow(['المرفقات (صور الفواتير)'], bold: true);
  for (var i = 0; i < photos.length; i++) {
    addRow([fmtDate(parseYmd(photos[i]['date'] as String)), 'فاتورة ${i + 1}']);
  }
  addRow(['']);

  addRow(['المسؤول المالي'], bold: true);
  addRow([username]);

  final bytes = excel.encode() ?? <int>[];
  if (bytes.isEmpty) throw StateError('الملف فارغ');

  if (!await dir.exists()) await dir.create(recursive: true);
  final file = File('${dir.path}/${reportFileName(from, to)}.xlsx');
  await file.writeAsBytes(bytes, flush: true);
  return file.path;
}

/// اشتقاق المفتاح من كلمة المرور (PBKDF2-HMAC-SHA256، 20000 تكرار).
/// يُستخدم لتشفير النسخ الاحتياطية، ولتخزين كلمة مرور الحساب بشكل آمن.
Uint8List deriveKey(String password, List<int> salt) {
  const iterations = 20000;
  final hmac = Hmac(sha256, utf8.encode(password));
  var block = hmac.convert([...salt, 0, 0, 0, 1]).bytes;
  final result = List<int>.from(block);
  for (var i = 1; i < iterations; i++) {
    block = hmac.convert(block).bytes;
    for (var j = 0; j < result.length; j++) {
      result[j] ^= block[j];
    }
  }
  return Uint8List.fromList(result);
}

/// سطر يوم واحد في الحساب.
class DayRow {
  const DayRow({
    required this.date,
    required this.start,
    required this.received,
    required this.spent,
    required this.end,
  });

  final String date;

  /// الرصيد بداية اليوم، بعد إضافة ما استُلم فيه.
  final int start;
  final int received;
  final int spent;

  /// الرصيد نهاية اليوم، وقد يكون سالباً.
  final int end;

  int get remaining => end > 0 ? end : 0;
  int get debt => end < 0 ? -end : 0;
}

class Ledger {
  Ledger._();

  /// يحسب الرصيد يوماً بيوم:
  /// بداية اليوم = رصيد السابق + المستلم، ونهايته = البداية - المصروف.
  /// الأيام التي لا حركة فيها لا تظهر، ويستمر الرصيد عبرها.
  /// إذا صار الرصيد سالباً فهذه مديونية، ويُسدَّد الدَّين أولاً من أي استلام لاحق.
  static List<DayRow> build() {
    final byDate = <String, List<Map<String, dynamic>>>{};
    for (final e in Store.all('entries')) {
      byDate.putIfAbsent(e['date'] as String, () => []).add(e);
    }
    final dates = byDate.keys.toList()..sort();

    final rows = <DayRow>[];
    var balance = 0;
    for (final d in dates) {
      final list = byDate[d]!;
      final received = sumOf(list.where((e) => e['kind'] == kindReceived));
      final spent = sumOf(list.where((e) => e['kind'] != kindReceived));
      final start = balance + received;
      final end = start - spent;
      rows.add(DayRow(
        date: d,
        start: start,
        received: received,
        spent: spent,
        end: end,
      ));
      balance = end;
    }
    return rows;
  }

  /// الرصيد الحالي: آخر نهاية يوم، أو صفر إذا لا توجد حركات.
  static int currentBalance(List<DayRow> rows) => rows.isEmpty ? 0 : rows.last.end;
}

/// طبقة التخزين: قاعدة Hive مشفّرة، ومفتاحها محفوظ في مخزن الجهاز الآمن.
class Store {
  Store._();

  static const _keyAlias = 'hisabati_db_key';
  static const backupKeys = ['entries', 'attachments'];
  static late Box<Map> _box;

  static Future<void> init() async {
    await Hive.initFlutter();
    const secure = FlutterSecureStorage();
    var encoded = await secure.read(key: _keyAlias);
    if (encoded == null) {
      final rnd = Random.secure();
      encoded = base64Encode(List<int>.generate(32, (_) => rnd.nextInt(256)));
      await secure.write(key: _keyAlias, value: encoded);
    }
    _box = await Hive.openBox<Map>(
      'hisabati_db',
      encryptionCipher: HiveAesCipher(base64Decode(encoded)),
    );
  }

  static ValueListenable<Box<Map>> get listenable => _box.listenable();

  static String newId() =>
      DateTime.now().microsecondsSinceEpoch.toRadixString(36) +
      Random().nextInt(9999).toString();

  // ---------- الإعدادات ----------
  static Map<String, dynamic> get _settings =>
      Map<String, dynamic>.from(_box.get('settings') ?? {});

  static T setting<T>(String key, T fallback) =>
      (_settings[key] as T?) ?? fallback;

  static Future<void> setSetting(String key, dynamic value) async {
    final s = _settings;
    s[key] = value;
    await _box.put('settings', s);
  }

  static ThemeMode get themeMode {
    switch (setting<String>('theme', 'dark')) {
      case 'light':
        return ThemeMode.light;
      case 'system':
        return ThemeMode.system;
      default:
        return ThemeMode.dark;
    }
  }

  // ---------- القوائم ----------
  static List<Map<String, dynamic>> all(String key) {
    final raw = _box.get(key);
    if (raw == null) return [];
    return (raw['items'] as List)
        .map((e) => Map<String, dynamic>.from(e as Map))
        .toList();
  }

  static Future<void> upsert(String key, Map<String, dynamic> item) async {
    final items = all(key);
    final i = items.indexWhere((e) => e['id'] == item['id']);
    if (i >= 0) {
      items[i] = item;
    } else {
      items.add(item);
    }
    await _box.put(key, {'items': items});
  }

  static Future<void> remove(String key, String id) async {
    final items = all(key)..removeWhere((e) => e['id'] == id);
    await _box.put(key, {'items': items});
  }

  static Future<void> wipe() => _box.clear();

  // ---------- النسخ الاحتياطي ----------
  static Map<String, dynamic> exportData() => {
        'version': 1,
        'settings': _settings,
        for (final key in backupKeys) key: all(key),
      };

  static Future<void> restore(Map<String, dynamic> data) async {
    await _box.clear();
    final settingsData = data['settings'];
    if (settingsData is Map) {
      await _box.put('settings', Map<String, dynamic>.from(settingsData));
    }
    for (final key in backupKeys) {
      final items = data[key];
      await _box.put(key, {'items': items is List ? items : []});
    }
  }
}

class BackupScreen extends StatelessWidget {

  void _message(BuildContext context, String text) {
    ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text(text)));
  }

  Future<void> _export(BuildContext context) async {
    final password = await _askPassword(
      context,
      title: 'كلمة مرور النسخة',
      confirm: true,
    );
    if (password == null) return;
    try {
      await Backup.exportFile(password);
    } catch (_) {
      if (context.mounted) _message(context, 'تعذّر إنشاء النسخة الاحتياطية');
    }
  }

  Future<void> _import(BuildContext context) async {
    final text = await Backup.pickFileText();
    if (text == null || !context.mounted) return;
    final password = await _askPassword(context, title: 'كلمة مرور النسخة');
    if (password == null || !context.mounted) return;

    Map<String, dynamic> data;
    try {
      final plain = Backup.decrypt(text, password);
      data = Map<String, dynamic>.from(jsonDecode(plain) as Map);
    } catch (_) {
      if (context.mounted) {
        _message(context, 'كلمة المرور غير صحيحة، أو الملف تالف');
      }
      return;
    }

    final ok = await showDialog<bool>(
      context: context,
      builder: (ctx) => AlertDialog(
        title: const Text('استعادة النسخة؟'),
        content: const Text('ستُستبدل البيانات الحالية ببيانات النسخة.'),

          TextButton(onPressed: () => Navigator.pop(ctx, false), child: const Text('إلغاء')),
          FilledButton(onPressed: () => Navigator.pop(ctx, true), child: const Text('استعادة')),
        ],
      ),
    );
    if (ok != true) return;

    await Store.restore(data);
    if (context.mounted) {
      _message(context, 'تمت استعادة البيانات بنجاح');
      Navigator.pop(context);
    }
  }
